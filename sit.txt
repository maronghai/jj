jj set a 1
# {"a": 1}

jj set a 1 b 2 
# {"a":1,"b":2}

jj set a.a 11 a.b 12
# {"a":{"a":11,"b": 12}}

jj set a $(jj set a1 11) b $(jj set b1 21)
# {"a":{"a1":"11"},"b":{"b1":"21"}}



jj set a 1 b $(jj set bb 22 --json) c $(jj set cc 33 --json)
# { "a": 1, "bb": 22, "cc": 33 }


jj push a b c
# ["a","b","c"]

jj push .a b c
# {"a":[{"b":"c"}]}

jj set a [] push a b c d e
